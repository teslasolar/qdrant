Code Examples

Load screens by single tag:

// Load all screens with a specific tag
async function loadScreensByTag(tag) {
  const screens = await window.templateLoader.loadScreensByTag(tag);
  console.log(`Found ${screens.length} screens with tag "${tag}"`);

  // Load first screen
  if (screens.length > 0) {
    await window.templateLoader.renderScreen(screens[0], 'hmi-container');
  }
}

// Usage
loadScreensByTag('overview');

Load screens by multiple tags:

// Find screens matching ALL tags
async function loadScreensByMultipleTags(tags) {
  await window.templateLoader.loadIndexes();
  const allScreens = window.templateLoader.indexes.screens.screens;

  const matching = allScreens.filter(screen =>
    tags.every(tag => screen.tags && screen.tags.includes(tag))
  );

  return matching;
}

// Usage
const screens = await loadScreensByMultipleTags(['overview', 'dashboard']);

Get all unique tags:

// Extract all unique tags from screens
async function getAllTags() {
  await window.templateLoader.loadIndexes();
  const screens = window.templateLoader.indexes.screens.screens;

  const tagSet = new Set();
  screens.forEach(screen => {
    if (screen.tags) {
      screen.tags.forEach(tag => tagSet.add(tag));
    }
  });

  return Array.from(tagSet).sort();
}

// Usage
const tags = await getAllTags();
console.log('Available tags:', tags);
